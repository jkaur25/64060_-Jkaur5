set.seed(123)
n <- nrow(scaled_data)
idx <- sample(1:n, n/2)

A <- scaled_data[idx, ]
B <- scaled_data[-idx, ]

agnes_A <- agnes(A, method = "ward")
clusters_A <- cutree(agnes_A, k = 4)
centroids <- aggregate(A, by=list(cluster=clusters_A), mean)

assign_to_centroid <- function(df, centers){
  apply(df, 1, function(x){
    dists <- apply(centers[,-1], 1, function(c) sum((x-c)^2))
    which.min(dists)
  })
}

B_assigned <- assign_to_centroid(B, centroids)

agnes_full <- agnes(scaled_data, method = "ward")
full_clusters <- cutree(agnes_full, 4)
full_B <- full_clusters[-idx]
